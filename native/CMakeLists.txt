cmake_minimum_required(VERSION 3.20)
project(dsh_pdf_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PDFium CONFIG REQUIRED)
add_executable(dsh-pdf-native pdfium-worker.cpp)
target_include_directories(dsh-pdf-native PRIVATE "${JSON_INCLUDE_DIR}")
target_link_libraries(dsh-pdf-native PRIVATE pdfium)
if(MSVC)
  target_compile_options(dsh-pdf-native PRIVATE /W4 /EHsc /utf-8)
else()
  target_compile_options(dsh-pdf-native PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(APPLE)
  set_target_properties(dsh-pdf-native PROPERTIES BUILD_RPATH "@loader_path")
elseif(UNIX)
  set_target_properties(dsh-pdf-native PROPERTIES BUILD_RPATH "$ORIGIN")
endif()
